### 修复工具和脚本

#### 1. 批量类名修复脚本
```python
#!/usr/bin/env python3
"""批量修复Test_case → TestCase"""
import os
import re

def fix_test_case_names(directory):
    for root, dirs, files in os.walk(directory):
        for file in files:
            if file.endswith('.py'):
                filepath = os.path.join(root, file)
                with open(filepath, 'r', encoding='utf-8') as f:
                    content = f.read()
                
                # 修复常见的Test_case问题
                content = re.sub(r'unittest\.Test_case', 'unittest.TestCase', content)
                content = re.sub(r'Test_case', 'TestCase', content)
                
                with open(filepath, 'w', encoding='utf-8') as f:
                    f.write(content)

if __name__ == "__main__":
    fix_test_case_names("tests/")
    print("Test_case修复完成")
```
