### P0-6: 依赖注入容器优化

**问题描述**: 当前依赖注入容器存在命名问题和接口定义不完整

**具体问题**:
- 服务注册机制不完善
- 生命周期管理缺失
- 接口定义不够抽象
- 循环依赖检测缺失

**修复方案**:
```python
class DIContainer:
    def __init__(self):
        self._services = {}
        self._instances = {}
        self._building = set()  # 循环依赖检测
    
    def register_singleton(self, interface: Type[T], implementation: Type[T]):
        """注册单例服务"""
        self._services[interface] = (implementation, True)
    
    def register_transient(self, interface: Type[T], implementation: Type[T]):
        """注册瞬时服务"""
        self._services[interface] = (implementation, False)
    
    def resolve(self, interface: Type[T]) -> T:
        """解析服务实例"""
        if interface in self._building:
            raise ValueError(f"Circular dependency detected for {interface}")
        
        if interface in self._instances:
            return self._instances[interface]
        
        self._building.add(interface)
        try:
            implementation, is_singleton = self._services[interface]
            instance = implementation()
            
            if is_singleton:
                self._instances[interface] = instance
            
            return instance
        finally:
            self._building.discard(interface)
```

**验收标准**:
- 支持单例和瞬时服务注册
- 具备循环依赖检测
- 完整的生命周期管理
- 清晰的接口定义

```python
# 统一异常体系
class SystemException(Exception):
    def __init__(self, message: str, error_code: str = None, details: Dict = None):
        super().__init__(message)
        self.error_code = error_code
        self.details = details or {}
        self.timestamp = datetime.now()

class DataAccessError(SystemException):
    pass

class IndicatorCalculationError(SystemException):
    pass

# 统一异常处理装饰器
def exception_handler(reraise: bool = True, default_return=None):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            try:
                return func(*args, **kwargs)
            except SystemException:
                raise  # 重新抛出系统异常
            except Exception as e:
                logger.error(f"Unexpected error in {func.__name__}: {e}", exc_info=True)
                if reraise:
                    raise SystemException(f"Unexpected error: {e}")
                return default_return
        return wrapper
    return decorator
```

```python
# 性能监控装饰器
def performance_monitor(metric_name: str = None):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            start_time = time.time()
            try:
                result = func(*args, **kwargs)
                execution_time = time.time() - start_time
                
                # 记录性能指标
                metrics.record_execution_time(
                    metric_name or func.__name__, 
                    execution_time
                )
                
                # 记录成功调用
                metrics.increment_counter(f"{func.__name__}_success")
                
                return result
            except Exception as e:
                # 记录失败调用
                metrics.increment_counter(f"{func.__name__}_error")
                raise
        return wrapper
    return decorator

# 结构化日志
class StructuredLogger:
    def __init__(self, name: str):
        self.logger = logging.getLogger(name)
        
    def info(self, message: str, **kwargs):
        self.logger.info(json.dumps({
            'message': message,
            'timestamp': datetime.now().isoformat(),
            'level': 'INFO',
            **kwargs
        }))
```
