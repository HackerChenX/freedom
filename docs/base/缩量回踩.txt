## 15 分钟吸筹信号（15 分钟）
V11:=3*SMA((C-LLV(L,55))/(HHV(H,55)-LLV(L,55))*100,5,1)-2*SMA(SMA((C-LLV(L,55))/(HHV(H,55)-
LLV(L,55))*100,5,1),3,1);
V12:=(EMA(V11,3)-REF(EMA(V11,3),1))/REF(EMA(V11,3),1)*100;
AA:=(EMA(V11,3)<=13) AND FILTER((EMA(V11,3)<=13),15);
BB:=(EMA(V11,3)<=13 AND V12>13) AND FILTER((EMA(V11,3)<=13 AND V12>13),10);
XG: COUNT(AA OR BB,N) >= 1;


## 连续缩量（日线）
最小缩量比 := 0.95;  // 缩量幅度阈值（0.95=95%，值越大要求缩量越明显）

// ====== 核心条件 ======
连续缩量阴线 := 
    COUNT(                                                                     // 统计符合条件的天数
        (VOL < REF(VOL, 1) * 最小缩量比) AND (CLOSE < OPEN), // 单日缩量阴线
        N
    ) = N;  // 达到指定天数

// 最终选股条件
选股结果 := 连续缩量阴线;

## 放量下跌（日线）（负向指标，满足条件则剔除）
放量下跌 := 
    CLOSE < REF(CLOSE, 1)    // 今日收盘价低于昨日收盘价（收跌）
    AND VOL > REF(VOL, 1);    // 今日成交量大于昨日成交量

// 排除过去5日内出现过放量下跌的股票
选股条件 := NOT(EXIST(放量下跌, N));

## 放量大阴线（日线）（负向指标，满足条件则剔除）
放量大阴线 := 
    (CLOSE < OPEN * 0.96)         // 阴线实体跌幅>4%（可调整）
    AND VOL > MA(VOL, 5) * 1.8;   // 成交量超过5日均量1.8倍（可调整）

// 排除5日内存在放量大阴线的股票：
NOT(EXIST(放量大阴线, N));


## KDJ_DEA 任一上移（日线）
RSV:=(CLOSE-LLV(LOW,9))/(HHV(HIGH,9)-LLV(LOW,9))*100;
K:=SMA(RSV,3,1);

DIFF:=EMA(CLOSE,12)-EMA(CLOSE,26);
