# L4核心服务层标准调用模式
from utils.container import container
from db.interfaces.data_access_interface import DataAccessInterface
from db.interfaces.cache_interface import ICacheService

class L4CoreService:
    def __init__(self):
        # 使用依赖注入获取L3服务 (100%可靠)
        self.data_access = container.resolve("DataAccessInterface")
        self.cache_service = container.resolve("ICacheService")
    
    def process_stock_analysis(self, code: str):
        """L4层业务逻辑示例"""
        # 1. 通过L3层获取数据 (83.3%高质量保证)
        stock_data = self.data_access.get_stock_data(code, "2024-01-01", "2024-12-31")
        
        # 2. 使用L3层缓存服务 (78.1%扩展性保证)
        cache_key = f"analysis_{code}"
        cached_result = self.cache_service.get(cache_key)
        
        if cached_result:
            return cached_result
        
        # 3. 执行L4层核心业务逻辑
        analysis_result = self._perform_analysis(stock_data)
        
        # 4. 缓存结果
        self.cache_service.set(cache_key, analysis_result, ttl=3600)
        
        return analysis_result
