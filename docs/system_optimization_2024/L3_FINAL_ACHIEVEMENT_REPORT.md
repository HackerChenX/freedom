```python
# L4核心服务层标准调用模式
from utils.container import container

class L4CoreService:
    def __init__(self):
        # 使用依赖注入获取L3服务 (100%可靠)
        self.data_access = container.resolve("DataAccessInterface")
        self.cache_service = container.resolve("ICacheService")
        self.query_optimizer = container.resolve("QueryOptimizationService")
        
    def process_stock_data(self, stock_code: str):
        # 标准数据访问模式
        data = self.data_access.get_stock_data(
            code=stock_code,
            start_date="2024-01-01",
            end_date="2024-12-31"
        )
        
        # 标准缓存使用模式
        cache_key = f"stock_data_{stock_code}"
        cached_result = self.cache_service.get(cache_key)
        
        if not cached_result:
            # 处理数据...
            result = self.process_data(data)
            self.cache_service.set(cache_key, result, ttl=3600)
        
        return cached_result or result
```
