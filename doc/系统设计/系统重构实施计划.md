```python
class SQLRepository:
    """SQL语句仓库"""
    
    def __init__(self):
        self._sqls = self._load_sqls()
    
    def _load_sqls(self) -> Dict[str, str]:
        """加载SQL语句"""
        return {
            'get_stock_data': """
                SELECT {fields}
                FROM stock_data
                WHERE code IN ({codes})
                  AND date BETWEEN {start_date} AND {end_date}
                ORDER BY code, date
            """,
            'get_indicator_data': """
                SELECT code, date, indicator_name, value
                FROM indicator_results
                WHERE indicator_name = {indicator}
                  AND code IN ({codes})
                  AND date = {date}
            """,
            # 更多SQL语句...
        }
    
    def get_sql(self, name: str) -> str:
        """获取SQL语句"""
        if name not in self._sqls:
            raise ValueError(f"SQL语句 '{name}' 不存在")
        return self._sqls[name]
    
    def format_sql(self, name: str, **params) -> str:
        """格式化SQL语句"""
        sql = self.get_sql(name)
        return sql.format(**params)
```

```python
class IndicatorRegistryV2:
    """新版指标注册系统"""
    
    def __init__(self):
        self._indicators: Dict[str, Type[BaseIndicator]] = {}
        self._dependencies: Dict[str, List[str]] = {}
        self._metadata: Dict[str, Dict] = {}
    
    def register(self, indicator_class: Type[BaseIndicator]) -> bool:
        """注册指标"""
        name = indicator_class.get_name()
        
        # 检查依赖
        dependencies = indicator_class.get_dependencies()
        for dep in dependencies:
            if dep not in self._indicators:
                raise ValueError(f"依赖指标 '{dep}' 未注册")
        
        # 注册指标
        self._indicators[name] = indicator_class
        self._dependencies[name] = dependencies
        self._metadata[name] = indicator_class.get_metadata()
        
        return True
    
    def get_indicator(self, name: str) -> Type[BaseIndicator]:
        """获取指标类"""
        if name not in self._indicators:
            raise ValueError(f"指标 '{name}' 未注册")
        return self._indicators[name]
    
    def get_calculation_order(self, indicators: List[str]) -> List[str]:
        """获取计算顺序"""
        # 实现拓扑排序
        return self._topological_sort(indicators)
```

```python
class CalculationEngine:
    """计算引擎"""
    
    def __init__(self, registry: IndicatorRegistryV2, 
                 cache_manager: CacheManager):
        self.registry = registry
        self.cache = cache_manager
        self.executor = ThreadPoolExecutor(max_workers=4)
    
    def calculate_batch(self, indicators: List[str], 
                       stock_data: StockData) -> Dict[str, IndicatorResult]:
        """批量计算指标"""
        # 获取计算顺序
        calc_order = self.registry.get_calculation_order(indicators)
        
        results = {}
        for indicator_name in calc_order:
            # 检查缓存
            cache_key = self._generate_cache_key(indicator_name, stock_data)
            cached_result = self.cache.get(cache_key)
            
            if cached_result:
                results[indicator_name] = cached_result
            else:
                # 计算指标
                result = self._calculate_indicator(indicator_name, stock_data, results)
                results[indicator_name] = result
                
                # 缓存结果
                self.cache.set(cache_key, result)
        
        return results
    
    def calculate_parallel(self, indicators: List[str], 
                          stock_data: StockData) -> Dict[str, IndicatorResult]:
        """并行计算指标"""
        # 分析依赖关系，确定可并行计算的指标组
        parallel_groups = self._analyze_parallel_groups(indicators)
        
        results = {}
        for group in parallel_groups:
            # 并行计算当前组的指标
            futures = []
            for indicator_name in group:
                future = self.executor.submit(
                    self._calculate_indicator, 
                    indicator_name, stock_data, results
                )
                futures.append((indicator_name, future))
            
            # 等待当前组完成
            for indicator_name, future in futures:
                results[indicator_name] = future.result()
        
        return results
```

```python
class StrategyParserV2:
    """新版策略解析器"""
    
    def __init__(self, validator: StrategyValidator):
        self.validator = validator
    
    def parse(self, config_path: str) -> Strategy:
        """解析策略配置"""
        try:
            # 加载配置文件
            config = self._load_config(config_path)
            
            # 验证配置
            self.validator.validate(config)
            
            # 解析策略
            strategy = self._parse_strategy(config)
            
            return strategy
            
        except Exception as e:
            raise StrategyParsingError(f"策略解析失败: {e}")
    
    def _parse_strategy(self, config: Dict) -> Strategy:
        """解析策略对象"""
        # 解析基本信息
        basic_info = self._parse_basic_info(config)
        
        # 解析条件
        conditions = self._parse_conditions(config.get('conditions', []))
        
        # 解析过滤器
        filters = self._parse_filters(config.get('filters', {}))
        
        return Strategy(
            name=basic_info['name'],
            description=basic_info['description'],
            conditions=conditions,
            filters=filters,
            metadata=config.get('metadata', {})
        )
    
    def _parse_conditions(self, conditions_config: List[Dict]) -> List[Condition]:
        """解析条件列表"""
        conditions = []
        for cond_config in conditions_config:
            condition = self._parse_single_condition(cond_config)
            conditions.append(condition)
        return conditions
```

#### 3.2 类型安全的条件评估器
```python
class ConditionEvaluator:
    """条件评估器"""
    
    def __init__(self, type_checker: TypeChecker):
        self.type_checker = type_checker
    
    def evaluate(self, condition: Condition, 
                indicator_results: Dict[str, Any]) -> bool:
        """评估单个条件"""
        try:
            # 获取指标值
            indicator_value = self._get_indicator_value(
                condition.indicator, indicator_results
            )
            
            # 类型检查和转换
            typed_value = self.type_checker.check_and_convert(
                indicator_value, condition.expected_type
            )
            
            # 执行比较
            return self._execute_comparison(
                typed_value, condition.operator, condition.value
            )
            
        except Exception as e:
            raise ConditionEvaluationError(f"条件评估失败: {e}")
    
    def evaluate_group(self, conditions: List[Condition], 
                      logic_operator: str,
                      indicator_results: Dict[str, Any]) -> bool:
        """评估条件组"""
        if not conditions:
            return True
        
        results = []
        for condition in conditions:
            result = self.evaluate(condition, indicator_results)
            results.append(result)
        
        if logic_operator.upper() == 'AND':
            return all(results)
        elif logic_operator.upper() == 'OR':
            return any(results)
        else:
            raise ValueError(f"不支持的逻辑操作符: {logic_operator}")
```
