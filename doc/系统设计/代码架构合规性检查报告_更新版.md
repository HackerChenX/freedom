from typing import Dict, List, Any, Optional
import pandas as pd

class DataAccessInterface(ABC):
    """统一数据访问接口"""
    
    @abstractmethod
    def get_stock_data(self, code: str, period: str, start_date: str, end_date: str) -> pd.DataFrame:
        """获取股票数据"""
        pass
    
    @abstractmethod
    def get_indicator_data(self, code: str, indicator: str, params: Dict) -> pd.DataFrame:
        """获取指标数据"""
        pass
    
    @abstractmethod
    def execute_query(self, query: str, params: Dict) -> pd.DataFrame:
        """执行查询"""
        pass

    """ClickHouse数据访问实现"""

        params = {
            'code': code,

            'end_date': end_date

# 新建 utils/dependency_injection.py
from typing import Dict, Any, Callable, TypeVar, Type
from abc import ABC

T = TypeVar('T')

class DIContainer:
    """依赖注入容器"""
    
    def __init__(self):
        self._services: Dict[Type, Any] = {}
        self._singletons: Dict[Type, Any] = {}
        self._factories: Dict[Type, Callable] = {}
    
    def register_singleton(self, interface: Type[T], implementation: Type[T]):
        """注册单例服务"""
        self._services[interface] = (implementation, True)
    
    def register_transient(self, interface: Type[T], implementation: Type[T]):
        """注册瞬时服务"""
        self._services[interface] = (implementation, False)
    
    def register_factory(self, interface: Type[T], factory: Callable[[], T]):
        """注册工厂方法"""
        self._factories[interface] = factory
    
    def get(self, interface: Type[T]) -> T:
        """获取服务实例"""
        if interface in self._singletons:
            return self._singletons[interface]
        
        if interface in self._factories:
            instance = self._factories[interface]()
            return instance
            
        if interface in self._services:
            implementation, is_singleton = self._services[interface]
            instance = implementation()
            
            if is_singleton:
                self._singletons[interface] = instance
                
            return instance
        
        raise ValueError(f"Service {interface} not registered")

# 全局容器实例
container = DIContainer()

# 新建 db/sql_manager.py
from typing import Dict, Optional
from enum import Enum

class QueryType(Enum):
    """查询类型枚举"""
    STOCK_DATA = "stock_data"
    INDICATOR_DATA = "indicator_data"
    STRATEGY_CONFIG = "strategy_config"
    INDUSTRY_LIST = "industry_list"

class SQLManager:
    """SQL查询管理器"""
    
    def __init__(self):
        self.queries = {
            QueryType.STOCK_DATA: """
                SELECT * FROM stock_info 
                WHERE code = %(code)s 
                AND date BETWEEN %(start_date)s AND %(end_date)s
                ORDER BY date
            """,
            QueryType.INDICATOR_DATA: """
                SELECT * FROM indicator_%(indicator)s 
                WHERE code = %(code)s 
                AND date BETWEEN %(start_date)s AND %(end_date)s
                ORDER BY date
            """,
            QueryType.STRATEGY_CONFIG: """
                SELECT config FROM strategy_definitions 
                WHERE strategy_id = %(strategy_id)s 
                LIMIT 1
            """,
            QueryType.INDUSTRY_LIST: """
                SELECT DISTINCT industry 
                FROM stock_info 
                WHERE industry IS NOT NULL 
                ORDER BY industry
            """
        }
    
    def get_query(self, query_type: QueryType) -> str:
        """获取查询语句"""
        return self.queries.get(query_type, "")
    
    def validate_params(self, query_type: QueryType, params: Dict) -> bool:
        """验证查询参数"""
        required_params = {
            QueryType.STOCK_DATA: ['code', 'start_date', 'end_date'],
            QueryType.INDICATOR_DATA: ['code', 'indicator', 'start_date', 'end_date'],
            QueryType.STRATEGY_CONFIG: ['strategy_id'],
            QueryType.INDUSTRY_LIST: []
        }
        
        required = required_params.get(query_type, [])
        return all(param in params for param in required)

from functools import wraps
from typing import Callable, Any
import logging

logger = logging.getLogger(__name__)

def exception_handler(
    default_return=None,
    reraise: bool = True,
    log_level: str = 'error'
):
    """统一异常处理装饰器"""
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs) -> Any:
            try:
                return func(*args, **kwargs)
            except Exception as e:
                log_func = getattr(logger, log_level.lower())
                log_func(f"Exception in {func.__name__}: {str(e)}", exc_info=True)
                
                if reraise:
                    raise
                return default_return
        return wrapper
    return decorator

# 使用示例

class UnifiedConfigManager:
    """统一配置管理器"""

        config = self.config_cache[section]
        if key is None:
            return config
        
        return config.get(key)
    
    def get_database_config(self) -> Dict[str, Any]:
        """获取数据库配置"""
        db_config = self.get_config('database')
        
        # 解密敏感信息
        if 'password' in db_config and db_config['password'].startswith('ENC:'):

        return db_config
    
    def _load_config(self, section: str):
        """加载配置文件"""
        config_file = self.config_dir / f"{section}.yaml"
        if not config_file.exists():
            raise FileNotFoundError(f"Config file {config_file} not found")
        
        with open(config_file, 'r', encoding='utf-8') as f:
            self.config_cache[section] = yaml.safe_load(f)
